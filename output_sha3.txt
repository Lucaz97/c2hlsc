    {
        t = ((uint64_t)(0x02 | (1 << 2))) << (ctx->byteIndex * 8);
    }

    ctx->u.s[ctx->wordIndex] ^= (ctx->saved ^ t);
    ctx->u.s[SHA3_KECCAK_SPONGE_WORDS - (ctx->capacityWords & (~SHA3_USE_KECCAK_FLAG)) - 1] ^= 0x8000000000000000ULL;
    keccakf(ctx->u.s);

    for (i = 0; i < SHA3_KECCAK_SPONGE_WORDS; i++)
    {
        if (produced >= outlen_bytes)
        {
            break;
        }
        lane = ctx->u.s[i];
        for (b = 0; b < 8; b++)
        {
            if (produced >= outlen_bytes)
            {
                break;
            }
            out[produced] = (uint8_t)(lane & 0xFF);
            lane >>= 8;
            produced++;
        }
    }
}

void sha3_Finalize(sha3_context *ctx, uint8_t *out, int outlen_bytes)
{
    sha3_Finalize_hls(ctx, out, outlen_bytes);
}
