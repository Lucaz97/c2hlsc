int main(void)
{
    // Example 32-bit values to test
    int32_t test_values[] = {
        123456789,
        -98765432,
        3329,
        65536,
        1234,
        -1234
    };

    int n = sizeof(test_values) / sizeof(test_values[0]);

    printf("Testing montgomery_reduce():\n\n");

    for (int i = 0; i < n; i++) {
        int32_t a = test_values[i];
        int16_t r = montgomery_reduce(a);

        printf("a = %d  -->  montgomery_reduce(a) = %d\n", a, r);
    }

    return 0;
}
