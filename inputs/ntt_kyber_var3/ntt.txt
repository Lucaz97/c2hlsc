void ntt(int16_t r[N])
{
    unsigned int len, start, j, k = 1;

    /* * Strictly align with the Baseline:
     * Removed the global center_reduce32 preprocessing for r[j].
     * The Baseline starts the calculation directly here.
     */

    for (len = 128; len >= 2; len >>= 1) {
        for (start = 0; start < N; start = j + len) {
            int16_t zeta = zetas[k++];

            for (j = start; j < start + len; j++) {
                butterfly_bounded(&r[j], &r[j + len], zeta);
            }
        }
    }
}
