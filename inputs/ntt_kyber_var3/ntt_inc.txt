/* * Retain this function for use elsewhere (e.g., INTT or post-polynomial addition),
 * but we MUST NOT call it within the core NTT loop below.
 */
static inline int16_t center_reduce32(int32_t a)
{
    int32_t t = ((int32_t)BARRETT_V * a + (1 << 25)) >> 26;
    int32_t r = a - t * (int32_t)Q;

    int32_t gt = (QHALF - r) >> 31;  
    r -= gt & (int32_t)Q;
    
    int32_t lt = (r + QHALF) >> 31;  
    r += lt & (int32_t)Q;

    return (int16_t)r;
}

/* * Strictly align with the Baseline: 
 * Removed the final center_reduce32, keeping the native right shift.
 */
static inline int16_t montgomery_reduce_bounded(int32_t a)
{
    int32_t t = (int16_t)((int16_t)a * QINV);
    int32_t u = a - t * (int32_t)Q;
    return (int16_t)(u >> 16); 
}

/* * Strictly align with the Baseline: 
 * Removed the pre-reduction of input parameters.
 */
static inline int16_t fqmul_bounded(int16_t a, int16_t b)
{
    return montgomery_reduce_bounded((int32_t)a * b);
}

/* * Strictly align with the Baseline: 
 * Allow sum and diff to grow naturally (lazy reduction) without forced modular reduction.
 */
static inline void butterfly_bounded(int16_t *a, int16_t *b, int16_t zeta)
{
    int16_t t = fqmul_bounded(zeta, *b);
    int16_t old_a = *a; // Cache the old value to prevent overwriting
    
    *b = old_a - t;
    *a = old_a + t;
}
