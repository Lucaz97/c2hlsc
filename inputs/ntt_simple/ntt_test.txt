long power(int base, int exp)
{
    long result = 1;
    if (exp > 0)
    {
        for (int i = 0; i < exp; i++)
        {
            result *= base;
        }
    }
    else if (exp < 0)
    {
        for (int i = 0; i < -exp; i++)
        {
            result *= base;
        }
        result = 1 / result;
    }
    return result;
}

void printVector(int matrix[N])
{
    for (int i = 0; i < N; i++)
    {
        printf("%d\t\t", matrix[i]);
    }
    printf("\n");
}

void printMatrix(int matrix[N][N])
{
    for (int i = 0; i < N; i++)
    {
        for (int j = 0; j < N; j++)
        {
            printf("%d\t\t", matrix[i][j]);
        }
        printf("\n");
    }
}
