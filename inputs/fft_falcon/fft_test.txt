/* Simple helper to print first few FFT outputs */
void print_preview(const fpr arr[], size_t n) {
    size_t limit = (n < 8) ? n : 8;
    for (size_t i = 0; i < limit; i++) {
        printf("  [%2zu] %.8f\n", i, arr[i]);
    }
}
