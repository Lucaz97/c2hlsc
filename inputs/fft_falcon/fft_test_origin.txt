/*
 * FFT KAT Verification Tool
 * 
 * Reads KAT vectors from a file and verifies the FFT implementation
 */



int parse_fpr_array(const char *str, fpr *arr, size_t max_len) {
    size_t count = 0;
    char buffer[65536];
    strncpy(buffer, str, sizeof(buffer) - 1);
    buffer[sizeof(buffer) - 1] = '\0';
    
    char *token = strtok(buffer, " \t\n");
    while (token && count < max_len) {
        arr[count++] = strtod(token, NULL);
        token = strtok(NULL, " \t\n");
    }
    
    return count;
}

int verify_roundtrip(unsigned test_num, unsigned logn, 
                           const fpr *input, size_t n) {
    fpr *work = (fpr*)malloc(n * sizeof(fpr));
    memcpy(work, input, n * sizeof(fpr));
    
    falcon_FFT(work, logn);
    falcon_iFFT(work, logn);
    
    int pass = 1;
    double max_error = 0.0;
    
    for (size_t i = 0; i < n; i++) {
        double error = fabs(work[i] - input[i]);
        if (error > max_error) max_error = error;
        if (error > EPSILON) pass = 0;
    }
    
    // Only print for failed tests or every 10th test
    if (!pass) {
        printf("  Test %3u: FAIL (max_error = %.3e)\n", test_num, max_error);
    } else if (test_num % 10 == 0) {
        printf("  Test %3u: PASS (max_error = %.3e)\n", test_num, max_error);
    }
    
    free(work);
    return pass;
}

int verify_polynomial_mult(unsigned test_num, unsigned logn,
                                  const fpr *poly_a, const fpr *poly_b,
                                  const fpr *expected_result, size_t n) {
    fpr *a_work = (fpr*)malloc(n * sizeof(fpr));
    fpr *b_work = (fpr*)malloc(n * sizeof(fpr));
    
    memcpy(a_work, poly_a, n * sizeof(fpr));
    memcpy(b_work, poly_b, n * sizeof(fpr));
    
    falcon_FFT(a_work, logn);
    falcon_FFT(b_work, logn);
    falcon_poly_mul_fft(a_work, b_work, logn);
    falcon_iFFT(a_work, logn);
    
    int pass = 1;
    double max_error = 0.0;
    
    for (size_t i = 0; i < n; i++) {
        double error = fabs(a_work[i] - expected_result[i]);
        if (error > max_error) max_error = error;
        if (error > EPSILON) pass = 0;
    }
    
    // Only print for failed tests or every 10th multiplication test
    if (!pass) {
        printf("  Mult Test %3u: FAIL (max_error = %.3e)\n", test_num, max_error);
    } else if (test_num % 10 == 0 || test_num >= 86) {
        printf("  Mult Test %3u: PASS (max_error = %.3e)\n", test_num, max_error);
    }
    
    free(a_work);
    free(b_work);
    return pass;
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        printf("Usage: %s <kat_file>\n", argv[0]);
        return 1;
    }
    
    FILE *fp = fopen(argv[1], "r");
    if (!fp) {
        printf("Error: Cannot open file %s\n", argv[1]);
        return 1;
    }
    
    printf("\n");
    printf("╔════════════════════════════════════════════════╗\n");
    printf("║   FFT KAT Verification Tool                   ║\n");
    printf("║   Processing test vectors...                  ║\n");
    printf("╚════════════════════════════════════════════════╝\n");
    printf("\n");
    
    char line[65536];
    unsigned test_count = 0;
    unsigned pass_count = 0;
    unsigned fail_count = 0;
    
    // State variables
    unsigned count = 0;
    unsigned logn = 0;
    size_t n = 0;
    fpr input[MAX_N], fft_output[MAX_N], ifft_output[MAX_N];
    fpr poly_a[MAX_N], poly_b[MAX_N], result[MAX_N];
    char description[256] = "";
    int is_mult_test = 0;
    
    while (fgets(line, sizeof(line), fp)) {
        // Skip comments and empty lines
        if (line[0] == '#' || line[0] == '\n') {
            if (strstr(line, "Multiplication")) {
                is_mult_test = 1;
            } else if (strstr(line, "Test Vector")) {
                is_mult_test = 0;
            }
            continue;
        }
        
        // Parse key-value pairs
        char key[256], value[65000];
        if (sscanf(line, "%s = %[^\n]", key, value) == 2) {
            if (strcmp(key, "count") == 0) {
                count = atoi(value);
            } else if (strcmp(key, "logn") == 0) {
                logn = atoi(value);
            } else if (strcmp(key, "n") == 0) {
                n = atoi(value);
            } else if (strcmp(key, "description") == 0) {
                strncpy(description, value, sizeof(description) - 1);
            } else if (strcmp(key, "input") == 0) {
                parse_fpr_array(value, input, MAX_N);
            } else if (strcmp(key, "ifft_output") == 0) {
                parse_fpr_array(value, ifft_output, MAX_N);
                // Verify this test
                test_count++;
                if (verify_roundtrip(count, logn, input, n)) {
                    pass_count++;
                } else {
                    fail_count++;
                }
            } else if (strcmp(key, "poly_a") == 0) {
                parse_fpr_array(value, poly_a, MAX_N);
            } else if (strcmp(key, "poly_b") == 0) {
                parse_fpr_array(value, poly_b, MAX_N);
            } else if (strcmp(key, "result") == 0 && is_mult_test) {
                parse_fpr_array(value, result, MAX_N);
                // Verify multiplication test
                test_count++;
                if (verify_polynomial_mult(count, logn, poly_a, poly_b, result, n)) {
                    pass_count++;
                } else {
                    fail_count++;
                }
            }
        }
    }
    
    fclose(fp);
    
    printf("\n");
    printf("Processing complete. Total tests verified: %u\n", test_count);
    printf("\n");
    printf("╔════════════════════════════════════════════════╗\n");
    printf("║              VERIFICATION SUMMARY              ║\n");
    printf("╠════════════════════════════════════════════════╣\n");
    printf("║  Total Tests:  %-3u                            ║\n", test_count);
    printf("║  Passed:       %-3u                            ║\n", pass_count);
    printf("║  Failed:       %-3u                            ║\n", fail_count);
    printf("║  Success Rate: %.1f%%                         ║\n", 
           test_count > 0 ? (100.0 * pass_count / test_count) : 0.0);
    printf("╚════════════════════════════════════════════════╝\n");
    
    if (fail_count == 0) {
        printf("\n✓ All KAT tests passed!\n\n");
        return 0;
    } else {
        printf("\n✗ Some KAT tests failed.\n\n");
        return 1;
    }
}
