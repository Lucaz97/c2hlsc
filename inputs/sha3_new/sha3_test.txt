int main(void)
{

#define OUTLEN 64

    /* Test input */
    const uint8_t msg[] = {
        'a', 'b', 'c'};

    uint8_t out[OUTLEN];

    /* Run SHAKE256 */
    shake256(out, OUTLEN, msg, sizeof(msg));

    /* Print result */
    printf("SHAKE256(\"abc\") output (%d bytes):\n", OUTLEN);
    for (int i = 0; i < OUTLEN; i++)
    {
        printf("%02x", out[i]);
        if ((i & 0x0F) == 0x0F)
            printf("\n");
    }
    if ((OUTLEN & 0x0F) != 0)
        printf("\n");

    return 0;
}
